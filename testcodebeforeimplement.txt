import React, { useRef, useState } from "react";
import {
  View,
  Text,
  StyleSheet,
  Pressable,
  Dimensions,
  Image,
  FlatList,
  NativeSyntheticEvent,
  NativeScrollEvent,
} from "react-native";
import { Ionicons } from "@expo/vector-icons";
import { LinearGradient } from "expo-linear-gradient";
import { router } from "expo-router";

const { width, height } = Dimensions.get("window");

type WelcomeSlide = {
  id: string;
  title: string;
  description: string;
  image: any;
  icon: keyof typeof Ionicons.glyphMap;
};

const slides: WelcomeSlide[] = [
  {
    id: "1",
    title: "Welcome to MediCare",
    description:
      "Your complete hospital management experience. Access healthcare services quickly, easily, and securely from your mobile device.",
    image: require("../assets/images/hospital-hero.jpg"),
    icon: "medical",
  },
  {
    id: "2",
    title: "Manage Your Healthcare",
    description:
      "Book appointments, view doctors, manage your medical information, and stay connected with your healthcare team.",
    image: require("../assets/images/hospital-hero.jpg"),
    icon: "calendar",
  },
  {
    id: "3",
    title: "Healthcare Made Simple",
    description:
      "Everything you need in one place. Get started today and enjoy a smarter, faster, and more convenient healthcare experience.",
    image: require("../assets/images/hospital-hero.jpg"),
    icon: "heart",
  },
];

export default function WelcomeScreen() {
  const [currentIndex, setCurrentIndex] = useState(0);
  const flatListRef = useRef<FlatList>(null);

  const handleScroll = (
    event: NativeSyntheticEvent<NativeScrollEvent>
  ) => {
    const offsetX = event.nativeEvent.contentOffset.x;
    const index = Math.round(offsetX / width);

    if (index !== currentIndex) {
      setCurrentIndex(index);
    }
  };

  const handleNext = () => {
    if (currentIndex < slides.length - 1) {
      flatListRef.current?.scrollToIndex({
        index: currentIndex + 1,
        animated: true,
      });
    } else {
      router.replace("/login");
    }
  };

  const handleSkip = () => {
    router.replace("/login");
  };

  const renderSlide = ({ item }: { item: WelcomeSlide }) => {
    return (
      <View style={styles.slide}>
        {/* Background Image */}
        <Image source={item.image} style={styles.backgroundImage} />

        {/* Dark overlay */}
        <LinearGradient
          colors={[
            "rgba(0,0,0,0.15)",
            "rgba(0,0,0,0.45)",
            "rgba(0,0,0,0.92)",
          ]}
          style={styles.overlay}
        />

        {/* Logo / Icon */}
        <View style={styles.topContainer}>
          <View style={styles.logoCircle}>
            <Ionicons name="medkit" size={30} color="#ffffff" />
          </View>

          <Text style={styles.logoText}>MediCare</Text>
        </View>

        {/* Content */}
        <View style={styles.content}>
          <View style={styles.iconContainer}>
            <Ionicons name={item.icon} size={34} color="#ffffff" />
          </View>

          <Text style={styles.title}>{item.title}</Text>

          <Text style={styles.description}>{item.description}</Text>
        </View>
      </View>
    );
  };

  return (
    <View style={styles.container}>
      <FlatList
        ref={flatListRef}
        data={slides}
        renderItem={renderSlide}
        keyExtractor={(item) => item.id}
        horizontal
        pagingEnabled
        showsHorizontalScrollIndicator={false}
        onScroll={handleScroll}
        scrollEventThrottle={16}
        bounces={false}
      />

      {/* Bottom Controls */}
      <View style={styles.bottomContainer}>
        {/* Pagination */}
        <View style={styles.pagination}>
          {slides.map((_, index) => (
            <View
              key={index}
              style={[
                styles.dot,
                currentIndex === index && styles.activeDot,
              ]}
            />
          ))}
        </View>

        {/* Buttons */}
        <View style={styles.buttonRow}>
          {currentIndex < slides.length - 1 ? (
            <Pressable onPress={handleSkip} style={styles.skipButton}>
              <Text style={styles.skipText}>Skip</Text>
            </Pressable>
          ) : (
            <View style={styles.skipButton} />
          )}

          <Pressable onPress={handleNext} style={styles.nextButton}>
            <LinearGradient
              colors={["#0EA5E9", "#0284C7"]}
              style={styles.nextGradient}
            >
              <Text style={styles.nextText}>
                {currentIndex === slides.length - 1
                  ? "Get Started"
                  : "Next"}
              </Text>

              <Ionicons
                name={
                  currentIndex === slides.length - 1
                    ? "checkmark"
                    : "arrow-forward"
                }
                size={22}
                color="#ffffff"
              />
            </LinearGradient>
          </Pressable>
        </View>
      </View>
    </View>
  );
}

  },

  },

  logoCircle: {

    borderRadius: 25,
    backgroundColor: "rgba(14,165,233,0.9)",
    justifyContent: "center",
    alignItems: "center",
  },

  logoText: {
    color: "#ffffff",
    fontSize: 23,
    fontWeight: "800",
    marginLeft: 12,
  },

  content: {
    position: "absolute",
    left: 25,

  },

  iconContainer: {

  },

});
